    **Problem:** DATA-11 requires assumptions to be shown in detailed mode before fetching. Currently, assumptions are only generated post-fetch by clean_and_merge in the Celery task. At the confirming_sources stage, the assumptions array is empty, so AssumptionsChecklist renders with nothing.

    **Solution:** Generate pre-fetch assumptions from the detected sources metadata when entering confirming_sources stage in detailed mode. These are the standard data transformation assumptions that the user can toggle before fetch begins.

    **1. In WorkspacePage.tsx, add a helper function `buildPrefetchAssumptions` before the handlers:**

    ```typescript
    function buildPrefetchAssumptions(sources: ParsedSource[]): AssumptionItem[] {
      const items: AssumptionItem[] = [];
      for (const src of sources) {
        if (src.source === "FRED") {
          // Standard FRED transformations
          items.push({
            key: `log_${src.series_id}`,
            label: `Log-transform ${src.display_name} (${src.series_id})`,
            recommended: true,
            confirmed: null,
          });
          items.push({
            key: `pct_${src.series_id}`,
            label: `Express ${src.display_name} as % change`,
            recommended: false,
            confirmed: null,
          });
        }
        if (src.source === "YAHOO") {
          items.push({
            key: `log_${src.series_id}`,
            label: `Log-transform ${src.display_name} price`,
            recommended: false,
            confirmed: null,
          });
          items.push({
            key: `returns_${src.series_id}`,
            label: `Use returns instead of price levels for ${src.display_name}`,
            recommended: true,
            confirmed: null,
          });
        }
      }
      // Common assumptions
      if (sources.length > 1) {
        items.push({
          key: "lag_1",
          label: "Include 1-period lag for independent variables",
          recommended: false,
          confirmed: null,
        });
      }
      return items;
    }
    ```
