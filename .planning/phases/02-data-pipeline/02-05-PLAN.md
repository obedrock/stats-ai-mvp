    1. Create backend/app/tasks/data_pipeline.py — Celery task following the existing analysis.py pattern:
       ```python
       import json
       import os
       from .celery_app import celery_app
       from app.services.fred_fetcher import fetch_fred_series
       from app.services.yahoo_fetcher import fetch_yahoo_series
       from app.services.data_cache import get_cached, set_cached
       from app.services.frequency_resolver import check_frequency_conflict, apply_resolution
       from app.services.data_pipeline import clean_and_merge
       import pandas as pd

       @celery_app.task(bind=True, name="fetch_data")
       def fetch_data(self, job_id: str, sources: list, date_range: dict, mode: str = "quick",
                      resolution: dict | None = None):
           """Fetch data from all sources, cache, clean, merge, and return preview."""
           dataframes = {}
           cache_keys = []

           for source_info in sources:
               source = source_info["source"]
               series_id = source_info["series_id"]
               start = date_range["start"]
               end = date_range["end"]
               cache_key = f"{source.lower()}:{series_id}:{start}:{end}"
               cache_keys.append(cache_key)

               # Check cache first (DATA-15)
               cached = get_cached(cache_key)
               if cached:
                   self.update_state(state="PROGRESS", meta={
                       "stage": "fetching_data", "job_id": job_id,
                       "sub_status": f"Cache hit: {source}: {series_id}"
                   })
                   dataframes[series_id] = pd.read_json(cached)
                   continue

               self.update_state(state="PROGRESS", meta={
                   "stage": "fetching_data", "job_id": job_id,
                   "sub_status": f"Fetching {source}: {series_id}..."
               })

               if source == "FRED":
                   df = fetch_fred_series(series_id, start, end)
               elif source == "YAHOO":
                   df = fetch_yahoo_series(series_id, start, end)
               else:
                   raise ValueError(f"Unknown source: {source}")

               # Cache with appropriate TTL (FRED=24h, Yahoo=1h per CLAUDE.md)
               ttl = 86400 if source == "FRED" else 3600
               set_cached(cache_key, df.to_json(), ttl)
               dataframes[series_id] = df

           # Check frequency conflict (DATA-09)
           if len(dataframes) > 1:
               conflict = check_frequency_conflict(dataframes)
               if conflict["has_conflict"] and resolution:
                   # Apply user-confirmed resolution
                   for series_id, df in dataframes.items():
                       dataframes[series_id] = apply_resolution(
                           df, resolution["target_frequency"], resolution["method"]
                       )
               elif conflict["has_conflict"] and not resolution:
                   # Return conflict to frontend — task pauses here
                   return {
                       "status": "frequency_conflict",
                       "job_id": job_id,
                       "conflict": conflict,
                       "cache_keys": cache_keys,
                   }

           # Clean and merge (DATA-05, 06, 07, 08, 10)
           self.update_state(state="PROGRESS", meta={
               "stage": "fetching_data", "job_id": job_id,
               "sub_status": "Cleaning and merging data..."
           })
           result = clean_and_merge(dataframes, mode=mode)

           return {
               "status": "data_ready",
               "job_id": job_id,
               "preview_rows": result["preview_rows"],
               "column_stats": result["column_stats"],
               "assumptions": result["assumptions"],
               "total_rows": result["total_rows"],
               "cache_keys": cache_keys,
           }
       ```

    2. Create backend/app/routers/data.py with these endpoints:

       **POST /parse-prompt** — Parse a natural language prompt via Claude (D-01, D-02, D-03):
       - Accept PromptSubmit body
       - Call `map_prompt_to_sources(payload.prompt)` in a thread executor (Claude call takes 1-3s, anti-pattern per RESEARCH.md)
       - For each returned source: validate FRED series IDs by calling `await validate_series(source["series_id"])` for FRED sources
       - Build ParsedSource objects with valid=True/False and suggestions
       - Return PromptParseResponse

       **POST /parse-prompt/override** — Re-validate an overridden series ID (DATA-02):
       - Accept SourceOverride body
       - Call `await validate_series(payload.series_id)` for FRED; for YAHOO, basic validation (non-empty)
       - Return updated ParsedSource

       **POST /upload** — Upload a file (DATA-12, DATA-13):
       - Accept UploadFile (from fastapi import UploadFile, File)
       - Read content with `await file.read()`
       - Call `parse_uploaded_file(content, file.filename)` in thread executor
       - Return UploadResult

       **POST /upload/confirm-mapping** — Confirm column mappings (DATA-14):
       - Accept ConfirmMapping body
       - Store mapping in memory/session (for now, return 200 OK acknowledgment)
       - Return {"status": "confirmed"}

       **POST /fetch** — Start data fetching (triggers Celery task):
       - Accept FetchRequest body
       - Create Job in DB with prompt, data_sources (JSON), analysis_mode
       - Enqueue `fetch_data.delay(job_id, sources, date_range, mode, resolution)`
       - Update job with celery_task_id
       - Return JobCreated

       **POST /resolve-frequency** — Accept frequency resolution choice (DATA-09):
       - Accept ResolutionChoice + job_id
       - Re-enqueue fetch_data task with the resolution choice
       - Return JobCreated (new task ID)

       **GET /preview/{job_id}** — Get data preview (DATA-16):
       - Look up job, check Celery task result
       - If task result has status "data_ready": return DataPreview
       - If task result has status "frequency_conflict": return FrequencyConflict response
       - Else: return current stage info

       All endpoints require authentication (use `Depends(get_current_user)` from existing auth).
